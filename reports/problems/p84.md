```rust
fn largest_rectangle_area(mut heights: Vec<i32>) -> i32 {
        let mut stack: Vec<(i32, i32)> = vec![(0, 0)];
        heights.push(0);
        let n = heights.len();
        let mut res = 0;
        for i in 0..n {
            let x1 = (i + 1) as i32;
            let y1 = heights[i];
            let mut x3 = x1;
            while let Some(&(x2, y2)) = stack.last() {
                if y2 > y1 {
                    stack.pop();
                    res = res.max((x1 - x2) * y2);
                    x3 = x2;
                } else {
                    stack.push((x3, y1));
                    break;
                }
            }
        }
        res
    }
```

```rust
proof fn lemma_area_eq_term(h: Seq<i32>, e: (i32, i32), p: int, i: int)
    requires
        e.0 as int - 1 == p,
        e.1 as int == min_range(h, p, i + 1),
    ensures area_at(e, i + 1) == term(h, p, i),
{
    reveal(stack_ok);
    reveal(f2_ok);
    reveal(wit);
    reveal(wit_at);
    let a = i + 1 + 1 - e.0 as int;
    let b = i - p + 1;
    assert(a == b);
    assert(a * (e.1 as int) == b * (e.1 as int));
    assert(area_at(e, i + 1) == a * (e.1 as int));
    assert(term(h, p, i) == b * min_range(h, p, i + 1));
}

proof fn lemma_term_le_area(h: Seq<i32>, st: Seq<(i32, i32)>, t: int, i: int, lo: int)
    requires
        0 <= t < st.len(),
        st[t].0 as int - 1 <= lo <= i,
        st[t].1 as int == min_range(h, lo, i + 1),
        0 <= st[t].1 as int,
    ensures term(h, lo, i) <= area_at(st[t], i + 1),
{
    reveal(stack_ok);
    reveal(f2_ok);
    reveal(wit);
    reveal(wit_at);
    let y = st[t].1 as int;
    let a = i - lo + 1;
    let b = i + 1 + 1 - st[t].0 as int;
    assert(a <= b);
    assert(a * y <= b * y) by (nonlinear_arith)
        requires a <= b, 0 <= y;
    assert(term(h, lo, i) == a * min_range(h, lo, i + 1));
    assert(area_at(st[t], i + 1) == b * y);
}

proof fn lemma_push_ok(h: Seq<i32>, st: Seq<(i32, i32)>, ci: int, e: (i32, i32))
    requires
        stack_ok(h, st, ci),
        ci < h.len(),
        e.1 as int == h[ci] as int,
        1 <= (e.0 as int) <= ci + 1,
        (st[st.len() as int - 1].0 as int) < (e.0 as int),
        (st[st.len() as int - 1].1 as int) <= (e.1 as int),
        0 <= (e.1 as int) <= 10000,
        e.0 as int == ci + 1 || min_range(h, e.0 as int - 1, ci) > e.1 as int,
    ensures
        stack_ok(h, st.push(e), ci + 1),
{
    reveal(stack_ok);
    reveal(f2_ok);
    reveal(wit);
    reveal(wit_at);
    let n = st.len() as int;
    let s2 = st.push(e);
    let p3 = e.0 as int - 1;
    assert(s2.len() as int == n + 1);
    assert(s2[n] == e);
    assert forall|t: int| 0 <= t < n implies s2[t] == st[t] by {}
    assert forall|t: int| 0 <= t < n implies (st[t].1 as int) <= (e.1 as int) by {
        lemma_chain(st, t, n - 1);
    }
    assert forall|t: int| 1 <= t < n implies (st[t].0 as int) <= ci by {}
    assert forall|t: int| 1 <= t < n implies #[trigger] f2_ok(h, s2, t, ci + 1) by {
        assert(s2[t] == st[t]);
        assert(f2_ok(h, st, t, ci));
        lemma_chain(st, t, n - 1);
        assert(st[t].0 as int - 1 < ci);
        lemma_min_range_append(h, st[t].0 as int - 1, ci);
        assert(s2[t].1 as int == min_range(h, s2[t].0 as int - 1, ci + 1));
        assert(f2_ok(h, s2, t, ci + 1));
    }
    assert(s2[n].1 as int == min_range(h, s2[n].0 as int - 1, ci + 1)) by {
        if p3 == ci {
            assert(min_range(h, ci, ci + 1) == h[ci] as int);
        } else {
            lemma_min_range_append(h, p3, ci);
        }
    }
}

proof fn lemma_push_wit(h: Seq<i32>, st: Seq<(i32, i32)>, ci: int, e: (i32, i32))
    requires
        stack_ok(h, st, ci),
        ci < h.len(),
        e.1 as int == h[ci] as int,
        1 <= (e.0 as int) <= ci + 1,
        (st[st.len() as int - 1].0 as int) < (e.0 as int),
        (st[st.len() as int - 1].1 as int) <= (e.1 as int),
        0 <= (e.1 as int),
        e.0 as int == ci + 1 || min_range(h, e.0 as int - 1, ci) > e.1 as int,
        wit(h, st, e.0 as int - 1, ci),
    ensures
        wit(h, st.push(e), ci + 1, ci + 1),
{
    reveal(stack_ok);
    reveal(f2_ok);
    reveal(wit);
    reveal(wit_at);
    let n = st.len() as int;
    let s2 = st.push(e);
    let p3 = e.0 as int - 1;
    let y1 = e.1 as int;
    assert(s2[n] == e);
    assert forall|lo: int| 0 <= lo < ci + 1 implies #[trigger] wit_at(h, s2, lo, ci + 1) by {
        if lo < p3 {
            assert(wit_at(h, st, lo, ci));
            let t0 = choose|t: int| 0 <= t < st.len() && st[t].0 as int - 1 <= lo && min_range(h, lo, ci) == st[t].1 as int;
            lemma_chain(st, t0, n - 1);
            assert(s2[t0] == st[t0]);
            assert(lo < ci);
            lemma_min_range_append(h, lo, ci);
            assert(0 <= t0 < s2.len() && s2[t0].0 as int - 1 <= lo && min_range(h, lo, ci + 1) == s2[t0].1 as int);
            assert(wit_at(h, s2, lo, ci + 1));
        } else {
            assert(s2[n].0 as int - 1 <= lo);
            if lo == ci {
                assert(min_range(h, ci, ci + 1) == h[ci] as int);
            } else {
                assert(p3 < ci);
                lemma_min_range_mono(h, p3, lo, ci);
                lemma_min_range_append(h, lo, ci);
            }
            assert(min_range(h, lo, ci + 1) == y1);
            assert(0 <= n < s2.len() && s2[n].0 as int - 1 <= lo && min_range(h, lo, ci + 1) == s2[n].1 as int);
            assert(wit_at(h, s2, lo, ci + 1));
        }
    }
}

proof fn lemma_closing(h: Seq<i32>, st: Seq<(i32, i32)>, ci: int, e: (i32, i32), res: int)
    requires
        stack_ok(h, st, ci),
        ci < h.len(),
        0 <= res,
        max_int(res, pm(st, ci)) == max_area_prefix(h, ci),
        stack_ok(h, st.push(e), ci + 1),
        wit(h, st.push(e), ci + 1, ci + 1),
    ensures
        max_int(res, pm(st.push(e), ci + 1)) == max_area_prefix(h, ci + 1),
{
    reveal(stack_ok);
    reveal(f2_ok);
    reveal(wit);
    reveal(wit_at);
    let s2 = st.push(e);
    let big_e = max_area_ending_at(h, 0, ci);
    lemma_prefix_step(h, ci);
    lemma_ending_nonneg(h, 0, ci);
    // every entry of s2 has area <= E
    assert forall|t: int| 0 <= t < s2.len() implies area_at(#[trigger] s2[t], ci + 1) <= big_e by {
        if t == 0 {
            assert(s2[0].1 == 0);
            assert(area_at(s2[0], ci + 1) == 0);
        } else {
            let p = s2[t].0 as int - 1;
            assert(f2_ok(h, s2, t, ci + 1));
            assert(s2[t].1 as int == min_range(h, p, ci + 1));
            lemma_area_eq_term(h, s2[t], p, ci);
            lemma_ending_ge_term(h, 0, p, ci);
        }
    }
    lemma_pm_le(s2, ci + 1, big_e);
    lemma_pm_push(st, e, ci + 1);
    assert forall|t: int| 0 <= t < st.len() implies 0 <= #[trigger] st[t].1 by {}
    lemma_pm_mono(st, ci);
    let b = max_int(res, pm(s2, ci + 1));
    assert forall|lo: int| 0 <= lo <= ci implies #[trigger] term(h, lo, ci) <= b by {
        assert(wit_at(h, s2, lo, ci + 1));
        let t0 = choose|t: int| 0 <= t < s2.len() && s2[t].0 as int - 1 <= lo && #[trigger] min_range(h, lo, ci + 1) == s2[t].1 as int;
        assert(0 <= s2[t0].1);
        lemma_term_le_area(h, s2, t0, ci, lo);
        lemma_pm_ge_entry(s2, ci + 1, t0);
    }
    lemma_ending_le(h, 0, ci, b);
    assert(max_int(res, pm(st, ci)) == max_area_prefix(h, ci));
    assert(pm(st, ci) <= pm(s2, ci + 1));
    assert(pm(s2, ci + 1) <= big_e);
    assert(max_area_prefix(h, ci + 1) == max_int(max_area_prefix(h, ci), big_e));
    assert(b == max_int(res, pm(s2, ci + 1)));
}

proof fn lemma_pop_step(h: Seq<i32>, st: Seq<(i32, i32)>, ci: int, x3: int, y1: int, res: int)

        stack_ok(h, st, ci),
        ci < h.len(),

        ({
            let st2 = st.subrange(0, st.len() as int - 1);
            let x2 = st[st.len() as int - 1].0 as int;

            &&& 1 <= x2 <= ci
            &&& (st2[st2.len() as int - 1].0 as int) < x2

            &&& (x2 == ci + 1 || min_range(h, x2 - 1, ci) > y1)

proof fn lemma_final(h: Seq<i32>, st: Seq<(i32, i32)>, n: int, res: int)
    requires
        n == h.len(),
        n >= 1,
        h[n - 1] == 0,
        stack_ok(h, st, n),
        0 <= res,
        max_int(res, pm(st, n)) == max_area_prefix(h, n),
    ensures
        res == max_area_prefix(h, h.len() as int),

    reveal(stack_ok);
    reveal(f2_ok);
    reveal(wit);
    reveal(wit_at);
    assert forall|t: int| 0 <= t < st.len() implies area_at(#[trigger] st[t], n) <= 0 by {
        if t >= 1 {
            assert(f2_ok(h, st, t, n));
            assert(st[t].1 as int == min_range(h, st[t].0 as int - 1, n));
            lemma_min_range_le_elem(h, st[t].0 as int - 1, n, n - 1);
            assert(st[t].1 as int == 0);
            assert((n + 1 - (st[t].0 as int)) * (st[t].1 as int) == 0) by (nonlinear_arith)
                requires st[t].1 as int == 0;
        } else {
            assert(st[0].1 == 0);
            assert((n + 1 - (st[0].0 as int)) * (st[0].1 as int) == 0) by (nonlinear_arith)

}

proof fn lemma_agree_min(s: Seq<i32>, t: Seq<i32>, lo: int, hi: int)

        }
    }
}

proof fn lemma_agree_ending(s: Seq<i32>, t: Seq<i32>, a: int, end: int)

    }
}

proof fn lemma_agree_prefix(s: Seq<i32>, t: Seq<i32>, k: int)
    requires
        0 <= k <= s.len(),
        k <= t.len(),
        forall|j: int| 0 <= j < k ==> s[j] == t[j],
    ensures
        max_area_prefix(s, k) == max_area_prefix(t, k),
    decreases k,
{
    if k > 0 {
        lemma_agree_prefix(s, t, k - 1);
        lemma_agree_ending(s, t, 0, k - 1);
        assert(max_area_prefix(s, k) == max_int(max_area_prefix(s, k - 1), max_area_ending_at(s, 0, k - 1)));
        assert(max_area_prefix(t, k) == max_int(max_area_prefix(t, k - 1), max_area_ending_at(t, 0, k - 1)));
    } else {
        assert(max_area_prefix(s, k) == 0);
        assert(max_area_prefix(t, k) == 0);
    }
}

proof fn lemma_prefix_nonneg(s: Seq<i32>, k: int)
    requires 0 <= k,
    ensures max_area_prefix(s, k) >= 0,
    decreases k,
{
    if k > 0 {
        lemma_prefix_nonneg(s, k - 1);
        assert(max_area_prefix(s, k) == max_int(max_area_prefix(s, k - 1), max_area_ending_at(s, 0, k - 1)));
    } else {
        assert(max_area_prefix(s, k) == 0);
    }
}

proof fn lemma_orig(orig: Seq<i32>, h: Seq<i32>, res: int)
    requires
        h == orig.push(0),
        h.len() >= 1,
        res == max_area_prefix(h, h.len() as int),
    ensures
        res == max_area_prefix(orig, orig.len() as int),
{
    let n = h.len() as int;
    assert(orig.len() as int == n - 1);
    lemma_prefix_step(h, n - 1);
    assert(forall|j: int| 0 <= j < n - 1 ==> orig[j] == h[j]);
    lemma_agree_prefix(orig, h, n - 1);
    lemma_prefix_nonneg(h, n - 1);
    assert forall|lo: int| 0 <= lo <= n - 1 implies #[trigger] term(h, lo, n - 1) <= 0 by {
        lemma_min_range_le_elem(h, lo, n, n - 1);
        assert(h[n - 1] == 0);
        assert((n - 1 - lo + 1) * min_range(h, lo, n) <= 0) by (nonlinear_arith)
            requires min_range(h, lo, n) <= 0, n - 1 - lo + 1 >= 0;
    }
    lemma_ending_le(h, 0, n - 1, 0);
}
struct Solution;
impl Solution {
    #[verifier::loop_isolation(false)]
    fn largest_rectangle_area(mut heights: Vec<i32>) -> (result: i32)
        requires pre(heights@),
        ensures post(heights@, result),
// ==== BODY (prover) ====
    {
            let ghost orig = heights@;
            let mut stack: Vec<(i32, i32)> = vec![(0, 0)];
            heights.push(0);
            proof {
                lemma_hb_push(orig, heights@);
                assert(stack@[0] == (0i32, 0i32));
                lemma_init_stack(heights@, stack@);
            }
            let n = heights.len();
            let mut res = 0;
            for i in 0..n
                invariant
                    0 <= i <= n,
                    n as int == heights@.len(),
                    hb(heights@),
                    stack_ok(heights@, stack@, i as int),
                    wit(heights@, stack@, i as int, i as int),
                    0 <= res,
                    max_int(res as int, pm(stack@, i as int)) == max_area_prefix(heights@, i as int),
            {
                let x1 = (i + 1) as i32;
                let y1 = heights[i];
                let mut x3 = x1;
                let ghost mut pushed = false;
                while let Some(&(x2, y2)) = stack.last()
                    invariant
                        i < n,
                        n as int == heights@.len(),
                        x1 as int == i as int + 1,
                        y1 as int == heights@[i as int],
                        hb(heights@),
                        !pushed,
                        1 <= (x3 as int) <= i as int + 1,
                        0 <= res,
                        !pushed ==> (stack_ok(heights@, stack@, i as int)
                            && (stack@[stack@.len() as int - 1].0 as int) < (x3 as int)
                            && wit(heights@, stack@, x3 as int - 1, i as int)
                            && (x3 as int == i as int + 1 || min_range(heights@, x3 as int - 1, i as int) > y1 as int)
                            && max_int(res as int, pm(stack@, i as int)) == max_area_prefix(heights@, i as int)),
                        pushed ==> stack_ok(heights@, stack@, i as int + 1),
                        pushed ==> wit(heights@, stack@, i as int + 1, i as int + 1),
                        pushed ==> max_int(res as int, pm(stack@, i as int + 1)) == max_area_prefix(heights@, i as int + 1),
                    decreases stack@.len(),
                {
                    if y2 > y1 {
                        proof {
                            let top = stack@.len() as int - 1;
                            assert(stack@[top].1 == y2);
                            assert(stack@[top].0 == x2);
                            lemma_pop_step(heights@, stack@, i as int, x3 as int, y1 as int, res as int);
                            assert(((x1 as int) - (x2 as int)) * (y2 as int) == area_at(stack@[stack@.len() as int - 1], i as int));
                            assert(1 <= (x1 as int) - (x2 as int) <= 100000);
                            assert(0 <= ((x1 as int) - (x2 as int)) * (y2 as int) <= 1000000000) by (nonlinear_arith)
                                requires 1 <= (x1 as int) - (x2 as int) <= 100000, 0 <= (y2 as int) <= 10000;
                        }
                        let ghost pre_st = stack@;
                        stack.pop();
                        res = res.max((x1 - x2) * y2);
                        x3 = x2;
                        proof {
                            assert(stack@ =~= pre_st.subrange(0, pre_st.len() as int - 1));
                        }
                    } else {
                        proof {
                            let top = stack@.len() as int - 1;
                            assert(stack@[top].1 == y2);
                            assert(stack@[top].0 == x2);
                            assert((stack@[top].0 as int) < (x3 as int));
                            assert((stack@[top].1 as int) <= (y1 as int));
                            lemma_break_step(heights@, stack@, i as int, (x3, y1), res as int);
                        }
                        stack.push((x3, y1));
                        proof { pushed = true; }
                        break;
                    }
                }
            }
            proof {
                lemma_final(heights@, stack@, n as int, res as int);
                lemma_orig(orig, heights@, res as int);
                assert(res as int == largest_rect_area(orig));
            }
            res
        }
}
} // verus!
```

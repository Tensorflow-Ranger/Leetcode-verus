Constraints:
- `1 <= sx, sy, tx, ty <= 10^9`

Original rustgym code (never modified):

```rust
fn reaching_points(sx: i32, sy: i32, mut tx: i32, mut ty: i32) -> bool {
        while sx < tx && sy < ty {
            if tx < ty {
                ty %= tx;
            } else {
                tx %= ty;
            }
        }
        sx == tx && sy <= ty && (ty - sy) % sx == 0 || sy == ty && sx <= tx && (tx - sx) % sy == 0
    }
```

            sx as int,
            sy as int,
            tx as int,
            ty as int,
            ((tx as int) + (ty as int)) as nat,
        )
    } else {
        true
    }
}
// ==== LEMMAS (prover) ====
proof fn lemma_mod_step(a: int, b: int)
    requires
        1 <= b,
        b <= a,
    ensures
        0 <= a % b,
        a % b < b,
        a % b <= a - b,
{
    vstd::arithmetic::div_mod::lemma_fundamental_div_mod(a, b);
    assert(a / b >= 1) by (nonlinear_arith)
        requires 1 <= b, b <= a, a == b * (a / b) + a % b, 0 <= a % b < b;
    assert(b * (a / b) >= b * 1) by (nonlinear_arith)
        requires a / b >= 1, b >= 1;
}

proof fn lemma_exit_equiv(sx: int, sy: int, tx: int, ty: int, fuel: nat)
    requires
        sx >= 1,
        sy >= 1,
        0 <= tx,
        0 <= ty,
        fuel as int >= tx + ty,
        !(sx < tx && sy < ty),
    ensures
        (sx == tx && sy <= ty && (ty - sy) % sx == 0 || sy == ty && sx <= tx && (tx - sx) % sy == 0)
        == reach_fuel(sx, sy, tx, ty, fuel),
{
    reveal_with_fuel(reach_fuel, 2);
    if tx < sx || ty < sy {
        assert(reach_fuel(sx, sy, tx, ty, fuel) == false);
        assert((sx == tx && sy <= ty && (ty - sy) % sx == 0 || sy == ty && sx <= tx && (tx - sx) % sy == 0) == false);
    } else if tx == sx && ty == sy {
        assert(reach_fuel(sx, sy, tx, ty, fuel) == true);
        assert(ty - sy == 0);
        assert(0int % sx == 0) by (nonlinear_arith) requires sx >= 1;
        assert((sx == tx && sy <= ty && (ty - sy) % sx == 0 || sy == ty && sx <= tx && (tx - sx) % sy == 0) == true);
    } else if tx == sx {
        assert(ty > sy);
        assert(fuel >= 1);
        assert(reach_fuel(sx, sy, tx, ty, fuel) == ((ty - sy) % sx == 0));
        assert((sx == tx && sy <= ty && (ty - sy) % sx == 0 || sy == ty && sx <= tx && (tx - sx) % sy == 0) == ((ty - sy) % sx == 0));
    } else {
        assert(ty == sy);
        assert(tx > sx);
        assert(fuel >= 1);
        assert(reach_fuel(sx, sy, tx, ty, fuel) == ((tx - sx) % sy == 0));
        assert((sx == tx && sy <= ty && (ty - sy) % sx == 0 || sy == ty && sx <= tx && (tx - sx) % sy == 0) == ((tx - sx) % sy == 0));
    }
}
struct Solution;
impl Solution {
    #[verifier::loop_isolation(false)]
    fn reaching_points(sx: i32, sy: i32, mut tx: i32, mut ty: i32) -> (result: bool)
        requires pre(sx, sy, tx, ty),
        ensures post(sx, sy, tx, ty, result),
// ==== BODY (prover) ====
    {
        let ghost tx0 = tx;
        let ghost ty0 = ty;
        let ghost mut fuel: nat = (tx0 as int + ty0 as int) as nat;
        while sx < tx && sy < ty
            invariant
                sx as int >= 1,
                sy as int >= 1,
                0 <= tx as int,
                0 <= ty as int,
                fuel as int >= tx as int + ty as int,
                reach_fuel(sx as int, sy as int, tx as int, ty as int, fuel) ==
                    reach_fuel(sx as int, sy as int, tx0 as int, ty0 as int, (tx0 as int + ty0 as int) as nat),
            decreases fuel,
        {
            let ghost old_tx = tx;
            let ghost old_ty = ty;
            let ghost old_fuel = fuel;
            if tx < ty {
                ty %= tx;
            } else {
                tx %= ty;
            }
            proof {
                reveal_with_fuel(reach_fuel, 1);
                if old_tx < old_ty {
                    lemma_mod_step(old_ty as int, old_tx as int);
                } else {
                    lemma_mod_step(old_tx as int, old_ty as int);
                }
                assert(reach_fuel(sx as int, sy as int, old_tx as int, old_ty as int, old_fuel) ==
                       reach_fuel(sx as int, sy as int, tx as int, ty as int, (old_fuel as int - 1) as nat));
                fuel = (old_fuel as int - 1) as nat;
            }
        }
        proof {
            lemma_exit_equiv(sx as int, sy as int, tx as int, ty as int, fuel);
        }
        assert(
            ((sx as int) == (tx as int) && (sy as int) <= (ty as int) && ((ty as int) - (sy as int)) % (sx as int) == 0
             || (sy as int) == (ty as int) && (sx as int) <= (tx as int) && ((tx as int) - (sx as int)) % (sy as int) == 0) ==
            reach_fuel(sx as int, sy as int, tx0 as int, ty0 as int, (tx0 as int + ty0 as int) as nat)
        );
        sx == tx && sy <= ty && (ty - sy) % sx == 0 || sy == ty && sx <= tx && (tx - sx) % sy == 0
    }
}
} // verus!

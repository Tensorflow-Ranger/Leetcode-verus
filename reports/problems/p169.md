## Problem

LeetCode difficulty: Easy. Full statement: `runs/p169/problem.md`.

Constraints:
- `n == nums.length`
- `1 <= n <= 5 * 10^4`
- `-10^9 <= nums[i] <= 10^9`
- The input is generated such that a majority element will exist in the array.

Original rustgym code (never modified):

```rust
fn majority_element(nums: Vec<i32>) -> i32 {
        let mut count = 0;
        let mut num: Option<i32> = None;
        for x in nums {
            if let Some(y) = num {
                if x == y {
                    count += 1;
                } else {
                    count -= 1;
                    if count == 0 {
                        num = None;
                    }
                }
            } else {
                num = Some(x);
                count += 1;
            }
        }
        num.unwrap()
    }
```

                        } else if z == x {
                            lemma_good_other_bound(s, y, c, x);
                        } else {
                            lemma_good_other_bound(s, y, c, z);
                        }
                    }

                    assert(good(s2, None, 0));
                } else {
                    assert(step_cand(cand, c, x) == Some(y));
                    assert(step_count(cand, c, x) == c - 1);

                    lemma_count_push(s, x, y);
                    lemma_good_cand_bound(s, y, c);
                    assert(2 * count(s2, y) <= s2.len() + (c - 1));

                    assert forall|z: i32| z != y ==> #[trigger] (2 * count(s2, z)) <= s2.len() - (c - 1) by {
                        if z != y {
                            lemma_count_push(s, x, z);
                            if z == x {
                                lemma_good_other_bound(s, y, c, x);
                            } else {
                                lemma_good_other_bound(s, y, c, z);
                            }
                        }
                    }

                    assert(good(s2, Some(y), c - 1));
                }
            }
        }
    }
}

proof fn lemma_good_post(s: Seq<i32>, cand: Option<i32>, c: int)
    requires
        pre(s),
        good(s, cand, c),
    ensures
        post_opt(s, cand),
{
    let i = choose|i: int| 0 <= i < s.len() && count(s, s[i]) > s.len() / 2;
    assert(0 <= i < s.len() && count(s, s[i]) > s.len() / 2);

    lemma_div_majority(s.len() as int, count(s, s[i]));
    assert(2 * count(s, s[i]) > s.len() as int);

    match cand {
        None => {
            assert(c == 0);
            assert(forall|z: i32| #[trigger] (2 * count(s, z)) <= s.len() as int);
            assert(2 * count(s, s[i]) <= s.len() as int);
            assert(false);
            assert(post_opt(s, None));
        }
        Some(y) => {
            assert(c > 0);

            if s[i] == y {
                assert(count(s, y) > s.len() / 2);
            } else {
                assert(forall|z: i32| z != y ==> #[trigger] (2 * count(s, z)) <= s.len() as int - c);
                assert(2 * count(s, s[i]) <= s.len() as int - c);
                assert(2 * count(s, s[i]) <= s.len() as int - 1);
                assert(false);
            }

            assert(post_opt(s, Some(y)));
        }
    }
}
struct Solution;
impl Solution {
    #[verifier::loop_isolation(false)]
    fn majority_element(nums: Vec<i32>) -> (result: i32)
        requires pre(nums@),
        ensures post(nums@, result),
// ==== BODY (prover) ====
    {
        let mut count = 0;
        let mut num: Option<i32> = None;

        let ghost all = nums@;
        assert(pre(all));

        for x in it: nums
            invariant
                it.seq() == all,
                0 <= it.index() as int <= all.len() as int,
                good(all.take(it.index() as int), num, count as int),
                count as int <= all.len() as int,
                it.index() as int == all.len() as int ==> good(all, num, count as int),
        {
            let ghost old_num = num;
            let ghost old_c = count as int;
            let ghost old_prefix = all.take(it.index() as int);

            if let Some(y) = num {
                if x == y {
                    count += 1;
                } else {
                    assert(count > 0);
                    count -= 1;
                    if count == 0 {
                        num = None;
                    }
                }
            } else {
                num = Some(x);
                count += 1;
            }

            proof {
                assert((it.index() as int) < all.len() as int);
                assert(x == it.seq()[it.index() as int]);
                assert(x == all[it.index() as int]);
                assert(all.take((it.index() as int) + 1) =~= old_prefix.push(x));

                assert(num == step_cand(old_num, old_c, x));
                assert(count as int == step_count(old_num, old_c, x));

                lemma_good_step(old_prefix, old_num, old_c, x);

                assert(good(all.take((it.index() as int) + 1), num, count as int));

                if (it.index() as int) + 1 == all.len() as int {
                    assert(all.take((it.index() as int) + 1) =~= all);
                    assert(good(all, num, count as int));
                }
            }
        }

        proof {
            assert(good(all, num, count as int));
            lemma_good_post(all, num, count as int);
            assert(num.is_some());
        }

        num.unwrap()
    }
}

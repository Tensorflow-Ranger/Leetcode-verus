fn number_of_arithmetic_slices(a: Vec<i32>) -> i32 {
        let n = a.len();
        let mut dp: HashMap<(usize, i64), usize> = HashMap::new();
        let mut res = 0;
        for i in 0..n {
            for j in 0..i {
                let diff = a[i] as i64 - a[j] as i64;
                let prev = *dp.entry((j, diff)).or_insert(1);
                res += prev - 1;
                *dp.entry((i, diff)).or_insert(1) += prev;
            }
        }
        res as i32
    }

// Using a single self-recursive function avoids mutual recursion.
spec fn E(a: Seq<i32>, i: int, d: int, j: int) -> int
    decreases j
{
    if j <= 0 {
        0
    } else {
        E(a, i, d, j - 1)
            + (if a[j - 1] as int == a[i] as int - d {
                   E(a, j - 1, d, j - 1) + 1
               } else {
                   0
               })
    }
}

// Number of AP-subsets of size >= 2 ending at index i with common difference d.
spec fn EndD(a: Seq<i32>, i: int, d: int) -> int {
    if i <= 0 { 0 } else { E(a, i, d, i) }
}

// g(a,i) = number of AP-subsets of size >= 3 with maximum index exactly i,
// decomposed by second-largest index j:  sum_{j<i} EndD(a,j,a[i]-a[j]).
spec fn gsum(a: Seq<i32>, i: int, j: int) -> int
    decreases j
{
    if j <= 0 {
        0
    } else {
        gsum(a, i, j - 1) + EndD(a, j - 1, a[i] as int - a[j - 1] as int)
    }
}

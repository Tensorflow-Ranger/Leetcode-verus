spec fn contains_value(s: Seq<i32>, x: i32) -> bool
    decreases s.len()
{
    if s.len() == 0 {
        false
    } else if s.last() == x {
        true
    } else {
        contains_value(s.drop_last(), x)
    }
}

spec fn all_i32_range(s: Seq<i32>) -> bool
    decreases s.len()
{
    if s.len() == 0 {
        true
    } else {
        let last = s.last();
        all_i32_range(s.drop_last())
            && (i32::MIN as int) <= (last as int)
            && (last as int) <= (i32::MAX as int)
    }
}

spec fn max_value(s: Seq<i32>) -> i32
    decreases s.len()
{
    if s.len() == 0 {
        i32::MIN
    } else {
        let rest = max_value(s.drop_last());
        let last = s.last();
        if last > rest {
            last
        } else {
            rest
        }
    }
}

spec fn count_distinct_greater(s: Seq<i32>, x: int) -> int

        0

            if contains_value(s.drop_last(), last) {
                rest
            } else {
                rest + 1
            }
        } else {
            rest
        }
    }
}

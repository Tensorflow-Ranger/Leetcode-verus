```rust
fn min_swap(a: Vec<i32>, b: Vec<i32>) -> i32 {
        let n = a.len();
        let mut keep = vec![n; n];
        let mut swap = vec![n; n];
        keep[0] = 0;
        swap[0] = 1;
        for i in 1..n {
            if a[i - 1] < a[i] && b[i - 1] < b[i] {
                keep[i] = keep[i - 1];
                swap[i] = swap[i - 1] + 1;
            }
            if a[i - 1] < b[i] && b[i - 1] < a[i] {
                keep[i] = keep[i].min(swap[i - 1]);
                swap[i] = swap[i].min(keep[i - 1] + 1);
            }
        }
        swap[n - 1].min(keep[n - 1]) as i32
    }
```

spec fn is_valid_swap(a: Seq<i32>, b: Seq<i32>, s: Seq<bool>) -> bool {
    &&& s.len() == a.len()
    &&& forall|i: int| 0 <= i && i + 1 < a.len() as int ==>
        (if #[trigger] s[i] { b[i] as int } else { a[i] as int }) <
        (if s[i+1] { b[i+1] as int } else { a[i+1] as int })
    &&& forall|i: int| 0 <= i && i + 1 < a.len() as int ==>
        (if #[trigger] s[i] { a[i] as int } else { b[i] as int }) <
        (if s[i+1] { a[i+1] as int } else { b[i+1] as int })
}

spec fn count_true(s: Seq<bool>) -> int decreases s.len() {
    if s.len() == 0 {
        0
    } else {
        (if s.last() { 1int } else { 0int }) + count_true(s.drop_last())
    }
}

spec fn pre(a: Seq<i32>, b: Seq<i32>) -> bool {
    &&& 2 <= a.len() as int
    &&& a.len() as int <= 100000
    &&& b.len() == a.len()
    &&& forall|i: int| 0 <= i && i < a.len() as int ==> 0 <= #[trigger] a[i] && a[i] <= 200000
    &&& forall|i: int| 0 <= i && i < b.len() as int ==> 0 <= #[trigger] b[i] && b[i] <= 200000
    &&& exists|s: Seq<bool>| #[trigger] is_valid_swap(a, b, s)
}

spec fn post(a: Seq<i32>, b: Seq<i32>, result: i32) -> bool {
    exists|s: Seq<bool>| 
        #[trigger] is_valid_swap(a, b, s) &&
        count_true(s) == result as int &&
        forall|t: Seq<bool>| 
            #[trigger] is_valid_swap(a, b, t) ==> count_true(t) >= result as int
}

fn distribute_candies(candies: Vec<i32>) -> i32 {
        let n = candies.len();
        let mut hs: HashSet<i32> = HashSet::new();
        for val in candies {
            hs.insert(val);
        }
        usize::min(n / 2, hs.len()) as i32
    }

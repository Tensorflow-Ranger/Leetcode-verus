```rust
fn next_greater_element(nums1: Vec<i32>, nums2: Vec<i32>) -> Vec<i32> {
        let mut stack: Vec<i32> = vec![];
        let mut hm: HashMap<i32, i32> = HashMap::new();
        for x in nums2 {
            while let Some(last) = stack.pop() {
                if last > x {
                    stack.push(last);
                    break;
                } else {
                    hm.insert(last, x);
                }
            }
            stack.push(x);
        }
        nums1.iter().map(|x| *hm.get(x).unwrap_or(&-1)).collect()
    }
```

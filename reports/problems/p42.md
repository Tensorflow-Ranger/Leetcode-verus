fn trap(height: Vec<i32>) -> i32 {
        let n = height.len();
        if n == 0 {
            return 0;
        }
        let mut l = 0;
        let mut r = n - 1;
        let mut level = 0;
        let mut res = 0;
        while l < r {
            if height[l] < height[r] {
                let lower = height[l];
                level = level.max(lower);
                res += level - lower;
                l += 1;
            } else {
                let lower = height[r];
                level = level.max(lower);
                res += level - lower;
                r -= 1;
            }
        }
        res
    }

            assert(seq_max(s.push(x)) == max_int(s[0] as int, x as int));
        } else {
            assert(seq_max(s) == max_int(s[0] as int, seq_max(s.skip(1))));
            assert(seq_max(s.push(x)) == max_int(s[0] as int, max_int(seq_max(s.skip(1)), x as int)));
            lemma_max_int_assoc(s[0] as int, seq_max(s.skip(1)), x as int);
        }
    }
}

proof fn lemma_water_at_bounds(s: Seq<i32>, i: int)
    requires
        0 <= i < s.len(),
        forall|k: int| 0 <= k < s.len() ==> 0 <= #[trigger] s[k] <= 100000,
    ensures 0 <= water_at(s, i) <= 100000
{
    let left = seq_max(s.take(i + 1));
    let right = seq_max(s.skip(i));
    lemma_seq_max_take_contains(s, i);
    lemma_seq_max_skip_contains(s, i);
    lemma_seq_max_take_le(s, i + 1, 100000);
    lemma_seq_max_skip_le(s, i, 100000);
    assert(s[i] as int <= left);
    assert(s[i] as int <= right);
    assert(left <= 100000);
    assert(right <= 100000);
    if left <= right {
        assert(min_int(left, right) == left);
        assert(water_at(s, i) == left - s[i] as int);
        assert(0 <= water_at(s, i) <= 100000);
    } else {
        assert(min_int(left, right) == right);
        assert(water_at(s, i) == right - s[i] as int);
        assert(0 <= water_at(s, i) <= 100000);
    }
}

proof fn lemma_left_step(s: Seq<i32>, l: int, r: int, level: int)
    requires
        0 <= l < r < s.len(),
        0 <= s[l] <= 100000,
        0 <= s[r] <= 100000,
        level == max_int(seq_max(s.take(l)), seq_max(s.skip(r + 1))),
        seq_max(s.skip(r + 1)) <= seq_max(s.take(l + 1)),
        seq_max(s.take(l)) <= seq_max(s.skip(r)),
        level <= max_int(s[l] as int, s[r] as int),
        s[l] < s[r],
        forall|i: int| 0 <= i < s.len() ==> 0 <= #[trigger] s[i] <= 100000,
    ensures
        water_at(s, l) == max_int(level, s[l] as int) - s[l] as int,
        max_int(level, s[l] as int) == max_int(seq_max(s.take(l + 1)), seq_max(s.skip(r + 1))),
        max_int(level, s[l] as int) <= 100000,
        seq_max(s.skip(r + 1)) <= seq_max(s.take(l + 2)),
        seq_max(s.take(l + 1)) <= seq_max(s.skip(r)),
        max_int(level, s[l] as int) <= max_int(s[l + 1] as int, s[r] as int)
{
    let li = seq_max(s.take(l));
    let ri = seq_max(s.skip(r + 1));
    let h = s[l] as int;
    let hr = s[r] as int;
    let li2 = seq_max(s.take(l + 1));

    lemma_seq_max_take_step(s, l);
    assert(li2 == max_int(li, h));

    assert(level == max_int(li, ri));
    lemma_max_int_reorder(li, ri, h);
    assert(max_int(max_int(li, ri), h) == max_int(max_int(li, h), ri));
    assert(max_int(level, h) == max_int(max_int(li, ri), h));
    assert(max_int(level, h) == max_int(li2, ri));
    assert(ri <= li2);
    lemma_max_int_absorb_right(li2, ri);
    assert(max_int(level, h) == li2);

    lemma_seq_max_take_le(s, l, 100000);
    assert(li <= 100000);
    lemma_max_int_le(li, h, 100000);
    assert(li2 <= 100000);
    assert(max_int(level, h) <= 100000);

    lemma_seq_max_take_monotone(s, l + 1, l + 2);
    assert(li2 <= seq_max(s.take(l + 2)));
    assert(ri <= seq_max(s.take(l + 2)));

    lemma_seq_max_skip_contains(s, r);
    assert(hr <= seq_max(s.skip(r)));
    assert(h < hr);
    assert(h <= seq_max(s.skip(r)));
    assert(li <= seq_max(s.skip(r)));
    lemma_max_int_le(li, h, seq_max(s.skip(r)));
    assert(li2 <= seq_max(s.skip(r)));

    lemma_seq_max_skip_monotone(s, l, r);
    assert(seq_max(s.skip(r)) <= seq_max(s.skip(l)));
    assert(li2 <= seq_max(s.skip(l)));
    assert(water_at(s, l) == min_int(li2, seq_max(s.skip(l))) - h);
    lemma_min_int_eq_left(li2, seq_max(s.skip(l)));
    assert(min_int(li2, seq_max(s.skip(l))) == li2);
    assert(water_at(s, l) == li2 - h);
    assert(water_at(s, l) == max_int(level, h) - h);

    lemma_max_int_absorb_left(h, hr);
    assert(max_int(h, hr) == hr);
    assert(level <= hr);
    lemma_max_int_le(level, h, hr);
    assert(max_int(level, h) <= hr);
    lemma_max_int_ge_right(s[l + 1] as int, hr);
    assert(hr <= max_int(s[l + 1] as int, hr));
    assert(max_int(level, h) <= max_int(s[l + 1] as int, hr));
}

proof fn lemma_right_step(s: Seq<i32>, l: int, r: int, level: int)
    requires
        0 <= l < r < s.len(),
        0 <= s[l] <= 100000,
        0 <= s[r] <= 100000,
        level == max_int(seq_max(s.take(l)), seq_max(s.skip(r + 1))),
        seq_max(s.skip(r + 1)) <= seq_max(s.take(l + 1)),
        seq_max(s.take(l)) <= seq_max(s.skip(r)),
        level <= max_int(s[l] as int, s[r] as int),
        s[r] <= s[l],
        forall|i: int| 0 <= i < s.len() ==> 0 <= #[trigger] s[i] <= 100000,
    ensures
        water_at(s, r) == max_int(level, s[r] as int) - s[r] as int,
        max_int(level, s[r] as int) == max_int(seq_max(s.take(l)), seq_max(s.skip(r))),
        max_int(level, s[r] as int) <= 100000,
        seq_max(s.skip(r)) <= seq_max(s.take(l + 1)),
        seq_max(s.take(l)) <= seq_max(s.skip(r - 1)),
        max_int(level, s[r] as int) <= max_int(s[l] as int, s[r - 1] as int)
{
    let li = seq_max(s.take(l));
    let ri = seq_max(s.skip(r + 1));
    let hl = s[l] as int;
    let hr = s[r] as int;
    let ri2 = seq_max(s.skip(r));

    lemma_seq_max_skip_step(s, r);
    assert(ri2 == max_int(hr, ri));

    assert(level == max_int(li, ri));
    lemma_max_int_assoc(li, ri, hr);
    assert(max_int(max_int(li, ri), hr) == max_int(li, max_int(ri, hr)));
    lemma_max_int_comm(ri, hr);
    assert(max_int(ri, hr) == max_int(hr, ri));
    assert(max_int(level, hr) == max_int(li, ri2));
    assert(li <= ri2);
    lemma_max_int_absorb_left(li, ri2);
    assert(max_int(level, hr) == ri2);

    lemma_seq_max_skip_le(s, r, 100000);
    assert(ri2 <= 100000);
    assert(max_int(level, hr) <= 100000);

    assert(ri <= seq_max(s.take(l + 1)));
    lemma_seq_max_take_contains(s, l);
    assert(hl <= seq_max(s.take(l + 1)));
    assert(hr <= hl);
    assert(hr <= seq_max(s.take(l + 1)));
    lemma_max_int_le(hr, ri, seq_max(s.take(l + 1)));
    assert(ri2 <= seq_max(s.take(l + 1)));

    lemma_seq_max_skip_monotone(s, r - 1, r);
    assert(seq_max(s.skip(r)) <= seq_max(s.skip(r - 1)));
    assert(li <= seq_max(s.skip(r - 1)));

    assert(ri2 <= seq_max(s.take(l + 1)));
    lemma_seq_max_take_monotone(s, l + 1, r + 1);
    assert(seq_max(s.take(l + 1)) <= seq_max(s.take(r + 1)));
    assert(ri2 <= seq_max(s.take(r + 1)));
    assert(water_at(s, r) == min_int(seq_max(s.take(r + 1)), ri2) - hr);
    lemma_min_int_eq_right(seq_max(s.take(r + 1)), ri2);
    assert(min_int(seq_max(s.take(r + 1)), ri2) == ri2);
    assert(water_at(s, r) == ri2 - hr);
    assert(water_at(s, r) == max_int(level, hr) - hr);

    lemma_max_int_absorb_right(hl, hr);
    assert(max_int(hl, hr) == hl);
    assert(level <= hl);
    lemma_max_int_le(level, hr, hl);
    assert(max_int(level, hr) <= hl);
    lemma_max_int_ge_left(hl, s[r - 1] as int);
    assert(hl <= max_int(hl, s[r - 1] as int));
    assert(max_int(level, hr) <= max_int(hl, s[r - 1] as int));
}

proof fn lemma_final_zero(s: Seq<i32>, l: int, level: int)
    requires
        0 <= l < s.len(),
        s[l] >= 0,
        level == max_int(seq_max(s.take(l)), seq_max(s.skip(l + 1))),
        level <= s[l] as int,
        forall|i: int| 0 <= i < s.len() ==> 0 <= #[trigger] s[i] <= 100000,
    ensures water_at(s, l) == 0
{
    let li = seq_max(s.take(l));
    let ri = seq_max(s.skip(l + 1));
    assert(level == max_int(li, ri));
    lemma_max_int_ge_left(li, ri);
    lemma_max_int_ge_right(li, ri);
    assert(li <= level);
    assert(ri <= level);
    assert(li <= s[l] as int);
    assert(ri <= s[l] as int);

    lemma_seq_max_take_step(s, l);
    assert(seq_max(s.take(l + 1)) == max_int(li, s[l] as int));
    lemma_max_int_absorb_left(li, s[l] as int);
    assert(max_int(li, s[l] as int) == s[l] as int);
    assert(seq_max(s.take(l + 1)) == s[l] as int);

    lemma_seq_max_skip_step(s, l);
    assert(seq_max(s.skip(l)) == max_int(s[l] as int, ri));
    lemma_max_int_absorb_right(s[l] as int, ri);
    assert(max_int(s[l] as int, ri) == s[l] as int);
    assert(seq_max(s.skip(l)) == s[l] as int);

    assert(water_at(s, l) == min_int(s[l] as int, s[l] as int) - s[l] as int);
    assert(min_int(s[l] as int, s[l] as int) == s[l] as int);
    assert(water_at(s, l) == 0);
}

proof fn lemma_total_suffix_aux(s: Seq<i32>, k: int)
    requires 0 <= k <= s.len()
    ensures total_water(s, k) + suffix_water(s, k) == total_water(s, s.len() as int)
    decreases s.len() - k
{
    if k == s.len() {
        assert(suffix_water(s, k) == 0);
    } else {
        lemma_total_suffix_aux(s, k + 1);
        assert(k + 1 > 0);
        assert(suffix_water(s, k) == water_at(s, k) + suffix_water(s, k + 1));
        assert(total_water(s, k + 1) == total_water(s, k) + water_at(s, k));
        assert(total_water(s, k) + suffix_water(s, k) ==
               total_water(s, k + 1) + suffix_water(s, k + 1));
    }
}
struct Solution;
impl Solution {
    #[verifier::loop_isolation(false)]
    fn trap(height: Vec<i32>) -> (result: i32)
        requires pre(height@),
        ensures post(height@, result),
// ==== BODY (prover) ====
    {
        let n = height.len();
        if n == 0 {
            return 0;
        }
        let mut l = 0;
        let mut r = n - 1;
        let mut level = 0;
        let mut res = 0;
        proof {
            assert(n as int == height@.len());
            assert(n as int <= 20000);
        }
        while l < r
            invariant
                0 <= l <= r,
                r < n,
                n as int == height@.len(),
                0 <= level as int <= 100000,
                level as int == max_int(seq_max(height@.take(l as int)), seq_max(height@.skip(r as int + 1))),
                seq_max(height@.skip(r as int + 1)) <= seq_max(height@.take(l as int + 1)),
                seq_max(height@.take(l as int)) <= seq_max(height@.skip(r as int)),
                level as int <= max_int(height@[l as int] as int, height@[r as int] as int),
                res as int == total_water(height@, l as int) + suffix_water(height@, r as int + 1),
                0 <= res as int,
                res as int <= (l as int + (n as int - 1 - r as int)) * 100000,
            decreases r - l,
        {
            if height[l] < height[r] {
                let lower = height[l];
                let ghost old_level = level;
                let ghost old_res = res;
                proof {
                    lemma_left_step(height@, l as int, r as int, level as int);
                }
                level = level.max(lower);
                proof {
                    assert(level as int == max_int(old_level as int, lower as int));
                    assert(water_at(height@, l as int) == level as int - lower as int);
                }
                proof {
                    lemma_water_at_bounds(height@, l as int);
                    assert(0 <= level as int - lower as int <= 100000);
                    assert(2000000000 <= i32::MAX as int);
                }
                res += level - lower;
                proof {
                    assert(res as int == old_res as int + (level as int - lower as int));
                    assert(res as int == total_water(height@, l as int) + suffix_water(height@, r as int + 1) + water_at(height@, l as int));
                    assert(total_water(height@, l as int + 1) == total_water(height@, l as int) + water_at(height@, l as int));
                    assert(res as int == total_water(height@, l as int + 1) + suffix_water(height@, r as int + 1));
                }
                l += 1;
            } else {
                let lower = height[r];
                let ghost old_level = level;
                let ghost old_res = res;
                let ghost old_r = r;
                proof {
                    lemma_right_step(height@, l as int, r as int, level as int);
                }
                level = level.max(lower);
                proof {
                    assert(level as int == max_int(old_level as int, lower as int));
                    assert(water_at(height@, r as int) == level as int - lower as int);
                }
                proof {
                    lemma_water_at_bounds(height@, r as int);
                    assert(0 <= level as int - lower as int <= 100000);
                    assert(2000000000 <= i32::MAX as int);
                }
                res += level - lower;
                proof {
                    assert(res as int == old_res as int + (level as int - lower as int));
                    assert(res as int == total_water(height@, l as int) + suffix_water(height@, r as int + 1) + water_at(height@, r as int));
                    assert(suffix_water(height@, r as int) == water_at(height@, r as int) + suffix_water(height@, r as int + 1));
                    assert(res as int == total_water(height@, l as int) + suffix_water(height@, r as int));
                }
                r -= 1;
                proof {
                    assert(r as int + 1 == old_r as int);
                }
            }
        }
        proof {
            assert(l == r);
            assert(level as int == max_int(seq_max(height@.take(l as int)), seq_max(height@.skip(l as int + 1))));
            assert(max_int(height@[l as int] as int, height@[r as int] as int) == height@[l as int] as int);
            assert(level as int <= height@[l as int] as int);
            lemma_final_zero(height@, l as int, level as int);
            assert(water_at(height@, l as int) == 0);
            assert(res as int == total_water(height@, l as int) + suffix_water(height@, l as int + 1));
            lemma_total_suffix_aux(height@, l as int + 1);
            assert(total_water(height@, l as int + 1) + suffix_water(height@, l as int + 1) == total_water(height@, height@.len() as int));
            assert(total_water(height@, l as int + 1) == total_water(height@, l as int) + water_at(height@, l as int));
            assert(total_water(height@, l as int + 1) == total_water(height@, l as int));
            assert(res as int == total_water(height@, height@.len() as int));
            assert(n as int == height@.len());
            assert(res as int == total_water(height@, n as int));
        }
        res
    }
}

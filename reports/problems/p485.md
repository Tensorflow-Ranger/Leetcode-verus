fn find_max_consecutive_ones(nums: Vec<i32>) -> i32 {
        let mut max = 0;
        let mut count = 0;
        for x in nums {
            if x == 1 {
                count += 1;
                max = i32::max(count, max);
            } else {
                count = 0;
            }
        }
        max
    }

proof fn lemma_imax_rotate(a: int, b: int, c: int)
    ensures imax(a, imax(b, c)) == imax(b, imax(a, c)),
{
    if a >= b {
        if b >= c {
            assert(imax(a, imax(b, c)) == a);
            assert(imax(b, imax(a, c)) == a);
        } else {
            if a >= c {
                assert(imax(a, imax(b, c)) == a);
                assert(imax(b, imax(a, c)) == a);
            } else {
                assert(imax(a, imax(b, c)) == c);
                assert(imax(b, imax(a, c)) == c);
            }
        }
    } else {
        if b >= c {
            assert(imax(a, imax(b, c)) == b);
            assert(imax(a, c) <= b);
            assert(imax(b, imax(a, c)) == b);

                        pow4(old_steps + 1) == 4 * pow4(old_steps);

                assert(is_power_of_four_depth(old_n as int, 15 - old_steps) ==
                       is_power_of_four_depth(n as int, 14 - old_steps));

                steps = old_steps + 1;

                assert(15 - steps == 14 - old_steps);
                assert(n0 as int == n as int * pow4(steps));
                assert(is_power_of_four_spec(n0 as int) <==>
                       is_power_of_four_depth(n as int, 15 - steps));
                assert(0 <= steps && steps <= 15);
            }
        }

        proof {
            assert(!((n as int) != 0 && (n as int) % 4 == 0));
            lemma_depth_exit(n as int, 15 - steps);
            assert(is_power_of_four_spec(n0 as int) <==>
                   is_power_of_four_depth(n as int, 15 - steps));
            assert(is_power_of_four_depth(n as int, 15 - steps) <==> n == 1);
            assert((n == 1) <==> is_power_of_four_spec(n0 as int));
        }

        n == 1
    }
}

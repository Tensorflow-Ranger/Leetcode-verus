fn contains_nearby_duplicate(nums: Vec<i32>, k: i32) -> bool {
        let mut hs: HashSet<i32> = HashSet::new();
        let n = nums.len();
        let k = k as usize;
        for i in 0..n {
            let x = nums[i];
            if hs.contains(&x) {
                return true;
            } else {
                hs.insert(x);
                if hs.len() > k {
                    hs.remove(&nums[i - k]);
                }
            }
        }
        false
    }

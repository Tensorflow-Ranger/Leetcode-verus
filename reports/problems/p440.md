```rust
fn find_kth_number(n: i32, k: i32) -> i32 {
        let n = n as i64;
        let mut prefix = 1;
        let mut k = k as i64 - 1;
        while k > 0 {
            let mut first = prefix;
            let mut last = prefix + 1;
            let mut count = 0;
            while first <= n {
                count += last.min(n + 1) - first;
                first *= 10;
                last *= 10;
            }
            if count <= k {
                k -= count;
                prefix += 1;
            } else {
                k -= 1;
                prefix *= 10;
            }
        }
        prefix as i32
    }
```

```rust
spec fn lex_less_seq(a: Seq<int>, b: Seq<int>) -> bool
    decreases a.len() + b.len()
{
    if a.len() == 0 {
        b.len() > 0
    } else if b.len() == 0 {
        false
    } else if a[0] < b[0] {
        true
    } else if a[0] > b[0] {
        false
    } else {
        lex_less_seq(a.skip(1), b.skip(1))
    }
}

// Lexicographic comparison of positive integers by their decimal representations.
spec fn lex_less(a: int, b: int) -> bool {
    lex_less_seq(digits(a), digits(b))
}

spec fn count_lex_less(n: int, r: int) -> int
    decreases n
{
    if n <= 0 {
        0
    } else {
        count_lex_less(n - 1, r) + (if lex_less(n, r) { 1int } else { 0int })
    }
}
```

fn count_digit_one(n: i32) -> i32 {
        let mut m = 1i64;
        let mut res = 0i64;
        let n = n as i64;
        while m <= n {
            let d = 10 * m;
            res += n / d * m + m.min(0.max(n % d - m + 1));
            m *= 10;
        }
        res as i32
    }

// Counts how many times digit 1 appears at decimal position `p`
// among all numbers 0..n, assuming p is a positive power of 10.
spec fn count_at_position(n: int, p: int) -> int {
    if p <= 0 {
        0
    } else {
        let cycle = 10 * p;
        let full = (n / cycle) * p;
        let r = n % cycle;
        let extra = if r < p {
            0
        } else if r - p + 1 < p {
            r - p + 1
        } else {
            p
        };
        full + extra
    }
}

// Total number of digit 1 occurrences in all non-negative integers <= n,
// for n <= 10^9.  We sum over all possible decimal positions.
spec fn total_ones_upto(n: int) -> int {

                assert(sum_powers_lt(n, 0) == 0);
                assert(count_at_position(n, 0) == 0);
                assert(sum_powers_lt(n, m) == sum_powers_lt(n, q) + count_at_position(n, q));
                assert(sum_powers_lt(n, m) == 0);
                lemma_sum_pow10_nonneg(k);
                assert(k * (n / 10) >= 0) by (nonlinear_arith)
                    requires k >= 1, 0 <= n, n / 10 >= 0;
            } else {
                assert(1 <= q);
                lemma_sum_powers_lt_bound(n, q, k - 1);
                lemma_count_at_position_bound(n, q);

                assert(sum_powers_lt(n, m) ==
                    sum_powers_lt(n, q) + count_at_position(n, q));
                assert(count_at_position(n, q) <= n / 10 + q);
                assert(sum_pow10_upto(k) ==
                    sum_pow10_upto(k - 1) + pow10(k - 1));
                assert(k * (n / 10) == (k - 1) * (n / 10) + n / 10) by (nonlinear_arith);
            }
        }
    }
}

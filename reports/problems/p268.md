## Problem

LeetCode difficulty: Easy. Full statement: `runs/p268/problem.md`.

Constraints:
- `n == nums.length`
- `1 <= n <= 10^4`
- `0 <= nums[i] <= n`
- All the numbers of `nums` are **unique**.

Original rustgym code (never modified):

```rust
fn missing_number(nums: Vec<i32>) -> i32 {
        let mut xor: i32 = 0;
        let n = nums.len();
        for n in 0..=n {
            xor ^= n as i32;
        }
        for n in nums {
            xor ^= n;
        }
        xor
    }
```

            lemma_remove_all_absent(p, x);
            assert(remove_all_int(p, x) =~= p);
            assert(remove_all_int(s, x) == remove_all_int(p, x));
            lemma_remove_all_len_distinct(p, x);
            assert(remove_all_int(p, x).len() as int == p.len() as int);
            assert(remove_all_int(s, x).len() as int == s.len() as int - 1);
        } else {
            assert(contains_int(s, x as int) <==> contains_int(p, x as int));
            lemma_remove_all_len_distinct(p, x);
            assert(remove_all_int(s, x) == remove_all_int(p, x).push(s.last()));
            assert(remove_all_int(s, x).len() as int ==
                   remove_all_int(p, x).len() as int + 1);
            if contains_int(p, x as int) {
                assert(remove_all_int(p, x).len() as int == p.len() as int - 1);
                assert(remove_all_int(s, x).len() as int == s.len() as int - 1);
            } else {
                assert(remove_all_int(p, x).len() as int == p.len() as int);
                assert(remove_all_int(s, x).len() as int == s.len() as int);
            }
        }
    }
}

proof fn lemma_remove_all_distinct(s: Seq<i32>, x: i32)
    requires distinct_int(s)
    ensures distinct_int(remove_all_int(s, x))
    decreases s.len()
{
    if s.len() == 0 {
    } else {

            assert(!contains_int(p, s.last() as int));

                assert(s.last() as int != x as int);
                assert(contains_int(p, s.last() as int));
                assert(false);

proof fn lemma_xor_push(s: Seq<i32>, x: i32)

proof fn lemma_remove_all_xor(s: Seq<i32>, x: i32)
    requires distinct_int(s), contains_int(s, x as int)
    ensures xor_seq(s) == xor_seq(remove_all_int(s, x)) ^ x
    decreases s.len()
{
    if s.len() == 0 {
    } else {
        let p = s.drop_last();
        lemma_distinct_drop_last(s);
        if s.last() == x {
            lemma_distinct_last_not_in_drop(s);
            assert(!contains_int(p, x as int));
            lemma_remove_all_absent(p, x);
            assert(remove_all_int(p, x) =~= p);
            assert(remove_all_int(s, x) == remove_all_int(p, x));
            assert(xor_seq(s) == xor_seq(p) ^ x);
            assert(xor_seq(remove_all_int(s, x)) == xor_seq(p));
        } else {
            assert(contains_int(p, x as int));
            lemma_remove_all_xor(p, x);
            assert(remove_all_int(s, x) == remove_all_int(p, x).push(s.last()));
            lemma_xor_push(remove_all_int(p, x), s.last());
            assert(xor_seq(remove_all_int(s, x)) ==
                   xor_seq(remove_all_int(p, x)) ^ s.last());
            assert(xor_seq(s) == xor_seq(p) ^ s.last());
            assert(xor_seq(remove_all_int(p, x)) == xor_seq(p) ^ x) by {
                lemma_xor_cancel(xor_seq(remove_all_int(p, x)), x);
            }
            lemma_xor_swap(xor_seq(p), x, s.last());
            assert(xor_seq(remove_all_int(s, x)) ^ x == xor_seq(s));
        }
    }
}

proof fn lemma_distinct_len_bound(s: Seq<i32>, hi: int)

            assert(false);
        }
        assert(s.len() as int <= hi + 1);
    } else {
        assert(hi as i32 as int == hi);
        let x = hi as i32;

            lemma_remove_all_len_distinct(s, x);

proof fn lemma_distinct_subset_complete(s: Seq<i32>, n: int)
    requires
        0 <= n <= 10001,
        s.len() as int == n,
        forall|i: int| 0 <= i < s.len() ==> 0 <= s[i] as int && (s[i] as int) < n,
        distinct_int(s),
    ensures forall|x: int| 0 <= x < n ==> contains_int(s, x)
    decreases n
{
    if n == 0 {
        assert forall|x: int| 0 <= x < n ==> contains_int(s, x) by {
        };
    } else {
        assert((n - 1) as i32 as int == n - 1);
        let x = (n - 1) as i32;
        if contains_int(s, x as int) {
            let t = remove_all_int(s, x);
            lemma_remove_all_len_distinct(s, x);
            assert(t.len() as int == n - 1);
            lemma_remove_all_distinct(s, x);
            assert(distinct_int(t));
            assert forall|i: int|
                0 <= i < t.len() ==> 0 <= t[i] as int && (t[i] as int) < n - 1
            by {
                if 0 <= i && i < t.len() {
                    let y = t[i] as int;
                    lemma_contains_index(t, i);
                    assert(contains_int(t, y));
                    lemma_remove_all_contains(s, x, y);
                    assert(y != n - 1 && contains_int(s, y));
                    lemma_contains_exists_index(s, y);
                    let j = choose|j: int| 0 <= j < s.len() && s[j] as int == y;
                    assert(0 <= s[j] as int && (s[j] as int) < n);
                    assert(0 <= y && y < n);
                    assert(y != n - 1);
                    assert(y < n - 1);
                }
            };
            lemma_distinct_subset_complete(t, n - 1);
            assert forall|z: int| 0 <= z < n ==> contains_int(s, z) by {

                        lemma_remove_all_contains(s, x, z);
                        assert(contains_int(s, z));
                    } else {
                        assert(z == n - 1);
                        assert(contains_int(s, x as int));
                        assert(contains_int(s, z));
                    }
                }
            };
        } else {
            assert forall|i: int|
                0 <= i < s.len() ==> 0 <= s[i] as int && s[i] as int <= n - 2

proof fn lemma_xor_missing(nums: Seq<i32>)
    requires pre_aux(nums)
    ensures post(nums, xor_seq(range_seq(nums.len() as int + 1)) ^ xor_seq(nums))
    decreases nums.len()
{
    let n = nums.len() as int;
    assert(n <= 10000);
    assert(distinct_int(nums));
    if contains_int(nums, n) {
        assert(n >= 1);
        let x = n as i32;
        assert(x as int == n);
        let t = remove_all_int(nums, x);
        lemma_remove_pre_aux(nums);
        assert(pre_aux(t));
        lemma_remove_all_len_distinct(nums, x);
        assert(t.len() as int == n - 1);
        assert(t.len() < nums.len());
        lemma_xor_missing(t);

        lemma_remove_all_xor(nums, x);
        assert(xor_seq(nums) == xor_seq(t) ^ x);
        lemma_xor_push(range_seq(n), x);
        assert(xor_seq(range_seq(n + 1)) == xor_seq(range_seq(n)) ^ x);
        assert(t.len() as int + 1 == n);

        let r = xor_seq(range_seq(n + 1)) ^ xor_seq(nums);
        let rt = xor_seq(range_seq(t.len() as int + 1)) ^ xor_seq(t);
        assert(range_seq(t.len() as int + 1) == range_seq(n));
        assert(r == rt) by {
            lemma_xor_common(xor_seq(range_seq(n)), x, xor_seq(t));
        }
        assert(post(t, rt));

        assert(0 <= r as int && r as int <= n) by {
            assert(0 <= rt as int && rt as int <= t.len() as int);
            assert(rt as int <= n - 1);
        };

        assert(!contains_int(nums, r as int)) by {
            if contains_int(nums, r as int) {
                assert(r as int != n);
                assert(r as int != x as int);
                lemma_remove_all_contains(nums, x, r as int);
                assert(contains_int(t, r as int));
                assert(!contains_int(t, rt as int));
                assert(r == rt);
                assert(false);
            }
        };

        assert forall|z: int| 0 <= z <= n && z != r as int ==> contains_int(nums, z) by {
            if 0 <= z && z <= n && z != r as int {
                if z == n {
                    assert(contains_int(nums, n));
                } else {
                    assert(z <= n - 1);
                    assert(0 <= z <= t.len() as int);
                    assert(z != rt as int);
                    assert(contains_int(t, z));
                    assert(z != x as int);
                    lemma_remove_all_contains(nums, x, z);
                    assert(contains_int(nums, z));
                }
            }
        };
        assert(post(nums, r));
    } else {
        let x = n as i32;
        assert(x as int == n);
        assert forall|i: int|
            0 <= i < nums.len() ==> 0 <= nums[i] as int && (nums[i] as int) < n
        by {
            if 0 <= i && i < nums.len() {
                let y = nums[i] as int;
                assert(0 <= y && y <= n);
                if y == n {
                    lemma_contains_index(nums, i);
                    assert(contains_int(nums, n));
                    assert(false);
                }
                assert(y < n);
            }
        };
        lemma_distinct_subset_complete(nums, n);
        assert(forall|z: int| 0 <= z < n ==> contains_int(nums, z));
        lemma_xor_of_complete(nums, n);
        assert(xor_seq(nums) == xor_seq(range_seq(n)));
        lemma_xor_push(range_seq(n), x);
        assert(xor_seq(range_seq(n + 1)) == xor_seq(range_seq(n)) ^ x);
        let r = xor_seq(range_seq(n + 1)) ^ xor_seq(nums);
        lemma_xor_cancel_left(xor_seq(range_seq(n)), x);
        assert(r == x);
        assert(r as int == n);
        assert(!contains_int(nums, r as int));
        assert forall|z: int| 0 <= z <= n && z != r as int ==> contains_int(nums, z) by {
            if 0 <= z && z <= n && z != r as int {
                assert(z < n);
                assert(contains_int(nums, z));
            }
        };
        assert(post(nums, r));
    }
}
struct Solution;
impl Solution {
    #[verifier::loop_isolation(false)]
    fn missing_number(nums: Vec<i32>) -> (result: i32)
        requires pre(nums@),
        ensures post(nums@, result),
// ==== BODY (prover) ====
    {
        let mut xor: i32 = 0;
        let n = nums.len();
        let ghost N = n as int;
        proof {
            lemma_pre_pre_aux(nums@);
            assert(N == nums@.len());
            assert(N <= 10000);
        }
        for n in 0..=n
            invariant
                n as int <= N + 1,
                xor == xor_seq(range_seq(n as int)),
        {
            xor ^= n as i32;
            proof {
                assert(n as int <= N);
                assert((n as int) as i32 == n as i32);
                lemma_xor_push(range_seq(n as int), n as i32);
                assert(xor_seq(range_seq(n as int + 1)) ==
                       xor_seq(range_seq(n as int)) ^ (n as i32));
                assert(xor == xor_seq(range_seq(n as int + 1)));
            }
        }
        let ghost first_xor = xor;
        assert(first_xor == xor_seq(range_seq(N + 1)));
        let ghost nums_seq = nums@;
        proof {
            assert(nums_seq == nums@);
            lemma_pre_pre_aux(nums_seq);
            assert(pre_aux(nums_seq));
            lemma_xor_zero(first_xor);
            assert(xor_seq(nums_seq.take(0)) == 0);
        }
        for n in it: nums
            invariant
                it.seq() == nums_seq,
                it.index() <= nums_seq.len() as int,
                xor == first_xor ^ xor_seq(nums_seq.take(it.index())),
        {
            assert(it.seq()[it.index()] == n);
            xor ^= n;
            proof {
                let k = it.index();
                assert(0 <= k < nums_seq.len() as int);
                assert(nums_seq[k] == n);
                lemma_take_push(nums_seq, k);
                assert(nums_seq.take(k + 1) =~= nums_seq.take(k).push(nums_seq[k]));
                lemma_xor_push(nums_seq.take(k), n);
                assert(xor_seq(nums_seq.take(k + 1)) ==
                       xor_seq(nums_seq.take(k)) ^ n);
                lemma_xor_assoc(first_xor, xor_seq(nums_seq.take(k)), n);
                assert(xor == first_xor ^ xor_seq(nums_seq.take(k + 1)));
            }
        }
        proof {
            lemma_take_all(nums_seq);
        }
        assert(xor == first_xor ^ xor_seq(nums_seq));
        assert(first_xor == xor_seq(range_seq(N + 1)));
        assert(N == nums_seq.len() as int);
        assert(xor == xor_seq(range_seq(nums_seq.len() as int + 1)) ^ xor_seq(nums_seq));
        proof {
            assert(nums_seq == nums@);
            lemma_xor_missing(nums_seq);
            assert(post(nums_seq, xor));
        }
        xor
    }
}
